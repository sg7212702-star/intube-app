SAFE GET ELEMENT - तेरी login.html में जो भी ID हो पकड़ लेगा const getEl = (...ids) => { for(let id of ids){ let el = document.getElementById(id); if(el) return el; } return null; }; const emailInput = getEl('email','emailInput','loginEmail','userEmail'); const passInput = getEl('password','pass','passwordInput','loginPassword','userPass'); const loginBtn = getEl('loginBtn','loginButton','btnLogin','signinBtn'); const signupBtn = getEl('signupBtn','signupButton','btnSignup','registerBtn','createBtn'); const forgotBtn = getEl('forgotBtn','forgot','forgotPass'); const msgBox = getEl('msg','message','errorMsg','statusMsg'); function toast(m, err=false){ if(msgBox){ msgBox.innerText = m; msgBox.style.color = err? '#ff4d6d' : '#22c55e'; msgBox.style.display = 'block'; } else { console.log(m); if(err) alert(m); } } // 3. ALREADY LOGIN CHECK - Direct index.html onAuthStateChanged(auth, async (user)=>{ if(user){ toast("Already Login ✓ Redirect..."); try{ const ref = doc(db,"users",user.uid); const snap = await getDoc(ref); if(!snap.exists()){ await setDoc(ref,{ uid: user.uid, email: user.email, name: user.email.split('@')[0], bio: "Bio - Edit से बदलेगा", avatar: "https://cdn-icons-png.flaticon.com/512/149/149071.png", createdAt: serverTimestamp() }); } }catch(e){} setTimeout(()=> location.href="index.html", 700); } }); // 4. LOGIN LOGIC if(loginBtn){ loginBtn.onclick = async ()=>{ const email = emailInput?.value.trim(); const pass = passInput?.value.trim(); if(!email ||!pass) return toast("Email Password डालो", true); loginBtn.disabled = true; loginBtn.innerText = "Logging..."; try{ await signInWithEmailAndPassword(auth, email, pass); toast("Login Success ✓"); location.href="index.html"; }catch(e){ toast(e.message, true); loginBtn.disabled = false; loginBtn.innerText = "Log In"; } }; } // 5. SIGNUP LOGIC if(signupBtn){ signupBtn.onclick = async ()=>{ const email = emailInput?.value.trim(); const pass = passInput?.value.trim(); if(!email ||!pass) return toast("Email Password डालो", true); if(pass.length < 6) return toast("Password कम से कम 6 Character", true); signupBtn.disabled = true; signupBtn.innerText = "Creating..."; try{ const cred = await createUserWithEmailAndPassword(auth, email, pass); const uid = cred.user.uid; // Create Firestore User await setDoc(doc(db,"users",uid),{ uid: uid, email: email, name: email.split('@')[0], bio: "Bio - Edit से बदलेगा", avatar: "https://cdn-icons-png.flaticon.com/512/149/149071.png", followers: 0, following: 0, posts: 0, createdAt: serverTimestamp() }); toast("Account Created ✓"); location.href="index.html"; }catch(e){ toast(e.message, true); signupBtn.disabled = false; signupBtn.innerText = "Create Account"; } }; } // 6. FORGOT PASSWORD if(forgotBtn){ forgotBtn.onclick = async ()=>{ const email = emailInput?.value.trim(); if(!email) return toast("Email डालो फिर Forgot दबाओ", true); try{ await sendPasswordResetEmail(auth, email); toast("Reset Link भेज दिया ✓ Email Check करो"); }catch(e){ toast(e.message, true); } }; } console.log("login.js Loaded - Final ✓");
